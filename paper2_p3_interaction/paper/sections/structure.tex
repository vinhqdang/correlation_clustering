\section{Twins, vertices outside the packing, and a normal form}\label{sec:structure}
By Lemma~\ref{lem:decomp} the whole difficulty of $\ell>0$ is the set $U$ of at most $\ell$ unpacked edits. This section proves what can be said about $U$ without any further assumption on the instance beyond maximality of $H$ (in the pair sense of Remark~\ref{rem:max}).

\paragraph{Conventions.} A \emph{partition} $\Pi$ of $V$ into \emph{clusters} has \emph{cost} equal to the number of its \emph{edits}: edges between different clusters (deletions) plus non-edges inside a cluster (insertions). Solutions $S$ and partitions correspond ($G\sd S$ is a cluster graph whose cliques are the clusters), so $\OPT(G)$ is the minimum cost and an \emph{optimum} is a partition of minimum cost. $\Pi(v)$ is the cluster of $v$. Unless said otherwise, $H$ is \emph{pair-maximal} (Remark~\ref{rem:max}: no induced $P_3$ of $G$ has all three pairs unpacked) and $R=V\setminus V(H)$; every pair with an end in $R$ is unpacked.

\subsection{Twins}
\begin{lemma}[Twins, strict form]\label{lem:twin}
Let $u\ne v$ be adjacent with $N[u]=N[v]$. For every partition $\Pi$ with $\Pi(u)\ne\Pi(v)$, one of the two partitions obtained by moving $u$ into $\Pi(v)$, respectively $v$ into $\Pi(u)$, has cost smaller than $\Pi$ by at least $1$; more precisely the two cost changes sum to exactly $-2$. Hence $u,v$ lie in one cluster in \emph{every} optimum, with no hypothesis on $H$.
\end{lemma}
\begin{proof}
For a vertex $y$ and a set $D\not\ni y$ let $f(y,D)=|D\setminus N(y)|-|D\cap N(y)|$. Put $D_u=\Pi(u)\setminus\{u\}$ and $D_v=\Pi(v)\setminus\{v\}$ (possibly empty); $D_u\not\ni v$, $D_v\not\ni u$, and neither contains $u$ or $v$. Moving $u$ out of $\Pi(u)$ changes the status of the pairs $uz$, $z\in D_u$: before, $uz$ is an edit iff it is a non-edge; after, iff it is an edge; the change is $-f(u,D_u)$. Moving $u$ into $\Pi(v)=D_v\cup\{v\}$ changes, for $z\in D_v\cup\{v\}$, the status from ``edit iff edge'' to ``edit iff non-edge''; the change is $f(u,D_v\cup\{v\})=f(u,D_v)-1$ because $uv\in E$. All other pairs keep their status. So
\[\Delta_1=-f(u,D_u)+f(u,D_v)-1,\qquad \Delta_2=-f(v,D_v)+f(v,D_u)-1 .\]
Since $N(u)\setminus\{v\}=N(v)\setminus\{u\}$ we have $f(u,D)=f(v,D)$ for all $D\subseteq V\setminus\{u,v\}$, hence $\Delta_1+\Delta_2=-2$. Thus $\min(\Delta_1,\Delta_2)\le-1$.
\end{proof}
This is the critical-clique property used in kernelizations of Cluster Editing (Guo~\cite{Guo09}; also stated in~\cite{BFK18}); the argument above is self-contained and shows that it is strict.

\subsection{Viability of an unpacked edit}
\begin{lemma}[Viability of an unpacked edit]\label{lem:viab}
Let $S$ be any solution, $U$ its unpacked part and $uv\in U$. Let $D(uv)$ be the set of $z\notin Q(u)\cup Q(v)\cup\{u,v\}$ such that $\{u,v,z\}$ is an induced $P_3$ of $G\sd\{uv\}$. Then for every $z\in D(uv)$ one of $uz,vz$ lies in $U$, and $|D(uv)|\le|U|-1$.
\end{lemma}
\begin{proof}
All three pairs of $\{u,v,z\}$ are unpacked ($z\notin Q(u)\cup Q(v)$). If neither $uz$ nor $vz$ is in $S$, the triple has in $G\sd S$ the same pattern as in $G\sd\{uv\}$, an induced $P_3$. Distinct $z$ give distinct pairs, all different from $uv$.
\end{proof}
This gives a kernel-style condition (deleting $uv$ needs fewer than $\ell$ common neighbours outside $Q(u)\cup Q(v)$) but no bound on the number of candidate pairs: all pendant edges are viable.

\subsection{The vertices outside \texorpdfstring{$V(H)$}{V(H)}}
\begin{lemma}[Structure of $R$]\label{lem:R}
Let $H$ be pair-maximal. For $r,s\in R$: if $rs\in E$ then $N[r]=N[s]$; if $rs\notin E$ then $N(r)\cap N(s)=\emptyset$. Consequently:
\begin{enumerate}
\item every component $X$ of $G[R]$ (a \emph{class}) is a clique all of whose vertices have the same closed neighbourhood $N[X]$ (they are true twins);
\item $A_X:=N(X)\setminus X$ is contained in $V(H)$;
\item the sets $A_X$, over the classes $X$, are pairwise disjoint.
\end{enumerate}
\end{lemma}
\begin{proof}
Every pair with an end in $R$ is unpacked. Let $rs\in E$ and $z\ne r,s$ be adjacent to exactly one of $r,s$. Then $\{r,s,z\}$ has exactly two edges, i.e.\ is an induced $P_3$ with all three pairs unpacked, contradicting pair-maximality; so $N[r]=N[s]$. If $rs\notin E$ and $z$ is a common neighbour, then $r\,z\,s$ is an induced $P_3$ whose pairs $rz,zs,rs$ all have an end in $R$, again a contradiction.

(1) Adjacent vertices of $R$ have equal closed neighbourhoods; along a connected component all closed neighbourhoods coincide, and then any two vertices $r,t$ of the component satisfy $r\in N[r]=N[t]$, so they are adjacent. (2) A neighbour of $x\in X$ lying in $R$ is in the component of $x$, hence in $X$. (3) If $a\in A_X\cap A_Y$ for classes $X\ne Y$, then $a$ is a common neighbour of some $x\in X$ and $y\in Y$, which are non-adjacent (different components of $G[R]$); this contradicts the second clause.
\end{proof}
For $x\in X$ and $a\in V(H)$ we thus have $xa\in E$ iff $a\in A_X$; in particular vertices of one class are twins towards $V(H)$ as well. (Lemma~\ref{lem:R} fails for vertex-maximal packings, see Proposition~\ref{prop:vertexmax}.)

\begin{lemma}[Exchange formula]\label{lem:exch}
Let $\Pi$ be any partition, $X$ a class with $X\subseteq C$ for a cluster $C$, $x=|X|$ and $a=|A_X\cap C|$. Let $\Pi'$ be obtained by moving $X$ out of $C$ into a cluster of its own. Then $\mathrm{cost}(\Pi')-\mathrm{cost}(\Pi)=x\,(x+2a-|C|)$.
\end{lemma}
\begin{proof}
Only the pairs between $X$ and $C\setminus X$ change status. By Lemma~\ref{lem:R}, a vertex of $X$ has exactly $a$ neighbours in $C\setminus X$, namely $A_X\cap C$. Before the move such a pair is an edit iff it is a non-edge: $x(|C|-x-a)$ edits. After the move it is an edit iff it is an edge: $xa$ edits. The difference is $xa-x(|C|-x-a)=x(x+2a-|C|)$.
\end{proof}

\begin{theorem}[C1: no edit inside $R$]\label{thm:C1}
Let $H$ be pair-maximal. In \emph{every} optimum $\Pi$, every cluster meets $R$ in the empty set or in exactly one whole class. Hence no edit of $\Pi$ has both endpoints in $R$, and for every class $X$ the cluster $\Pi(X)$ equals $X\cup B_X$ with $B_X:=\Pi(X)\cap V(H)$.
\end{theorem}
\begin{proof}
By Lemma~\ref{lem:R}(1) and Lemma~\ref{lem:twin}, every class lies inside one cluster (a class of size $1$ trivially). Suppose a cluster $C$ contains two distinct classes $X_1,X_2$. Put $x_i=|X_i|\ge1$ and $a_i=|A_{X_i}\cap C|$. The sets $X_1,X_2\subseteq R$ and $A_{X_1}\cap C,A_{X_2}\cap C\subseteq V(H)$ are pairwise disjoint (Lemma~\ref{lem:R}(2),(3)), so $|C|\ge x_1+x_2+a_1+a_2$. Let $\Delta_i$ be the cost change of moving $X_i$ out of $C$. Since $\Pi$ is an optimum, $\Delta_1\ge0$ and $\Delta_2\ge0$, i.e.\ by Lemma~\ref{lem:exch} $|C|\le x_i+2a_i$ for $i=1,2$. With the lower bound on $|C|$: $x_2+a_2\le a_1$ and $x_1+a_1\le a_2$. Adding gives $x_1+x_2\le0$, absurd. So no cluster contains two classes.

An edit with both ends in $R$ joins two vertices $r,s$. If they lie in one class, they are adjacent and (by the above) in one cluster, so $rs$ is no edit. If they lie in different classes they are non-adjacent, and $rs$ is an edit iff they share a cluster, which was excluded.
\end{proof}
The strictness of the inequality is what makes the statement hold for every optimum (no ties, and no iteration or termination argument).

\begin{proposition}[Pair-maximality cannot be replaced by vertex-maximality]\label{prop:vertexmax}
Let $G$ have vertices $0,\dots,4$ and edges $03,13,14,24,34$, and $H=\{(0,3,1)\}$ (pairs $03,13,01$). Then $H$ is maximal among vertex-disjoint packings (only the two vertices $2,4$ remain), $R=\{2,4\}$, $\OPT=2$ and the unique optimum is $\{03,24\}$; it contains the edit $24$ with both endpoints in $R$. So Theorem~\ref{thm:C1} fails. Here $\{3,4,2\}$ is an induced $P_3$ all of whose pairs are unpacked, so $H$ is not pair-maximal and Lemma~\ref{lem:R} fails.
\end{proposition}
\begin{proof}
The partition $\{\{0\},\{1,3,4\},\{2\}\}$ has the two edits $03,24$. Conversely, $0\,3\,1$ and $3\,4\,2$ are induced $P_3$'s with disjoint pair sets $\{03,13,01\}$ and $\{34,24,23\}$, so $\OPT\ge2$. That this is the only optimum was checked by enumerating all $52$ partitions of the five vertices (computer; not re-derived by hand).
\end{proof}

\subsection{Consequences}\label{sec:conseq}
Let $(G,H,\ell)$ be a \emph{YES instance} with $H$ pair-maximal, $\Pi$ any optimum, $S$ its edit set, so $|S|=\OPT\le|H|+\ell$. Let $U$ be the unpacked edits. By Lemma~\ref{lem:decomp}, $|S|-|H|=\sum_i(|x_i|-1)+|U|$, so $|U|\le\ell$.  Write $q=\max_u|Q(u)|\le 2c$.

\begin{corollary}[Normal form]\label{cor:nf}
For every class $X$ let $B_X=\Pi(X)\cap V(H)$.
\begin{enumerate}
\item[(N1)] \emph{Types of unpacked edits.} Every unpacked edit is either an \emph{HH-edit} (both ends in $V(H)$) or an \emph{RH-edit} $\{x,a\}$, $x\in X$ a class, $a\in V(H)$; the latter is a deletion if $a\in A_X\setminus B_X$ and an insertion if $a\in B_X\setminus A_X$, and $\{x,a\}$ is edited iff $\{x',a\}$ is, for all $x,x'\in X$. With $h$ the number of HH-edits,
$|U|=h+\sum_X|X|\cdot|A_X\triangle B_X|\le\ell-\sum_i(|x_i|-1)\le \ell.$
\item[(N2)] \emph{Frozen classes.} If $|X|>\ell$, then $\Pi(X)=X\cup A_X$.
\item[(N3)] \emph{Attachment.} Every $a\in V(H)$ lies in $A_X$ for at most one class $X$. If $z,z'\in A_X$ are non-adjacent then $zz'$ is packed. If $a\in V(H)\setminus A_X$, $z\in A_X$ and $az\in E$ then $az$ is packed; hence $|N(a)\cap A_X|\le|Q(a)|\le q$.
\item[(N4)] \emph{Shrink rule.} If $|X|\ge\ell+2$ and $x\in X$, then $(G,H,\ell)$ is a YES instance iff $(G-x,H,\ell)$ is. This holds for $H$ pair-maximal without the YES hypothesis. If $A_X=\emptyset$ then $X$ is a component of $G$ and may be deleted without changing $\OPT$.
\item[(N5)] \emph{Size.} After exhausting the two rules in (N4), every class has at most $\ell+1$ vertices and at most $|V(H)|$ classes have $A_X\ne\emptyset$, so $|R|\le(\ell+1)|V(H)|$ and $n\le(\ell+2)|V(H)|\le 3(\ell+2)|H|$.
\end{enumerate}
\end{corollary}
\begin{proof}
(N1) Pairs with an end in $R$ are unpacked; by Theorem~\ref{thm:C1} none has both ends in $R$. The vertices of $X$ are twins towards $V(H)$ and lie in one cluster, which gives the all-or-none statement; $xa$ is an edit iff exactly one of $a\in B_X$, $a\in A_X$ holds. Counting gives the identity; the bound is the first paragraph of this subsection.

(N2) If $\Pi(X)\ne X\cup A_X$ then, since $\Pi(X)=X\cup B_X$ (Theorem~\ref{thm:C1}), $A_X\triangle B_X\neq\emptyset$ and (N1) gives $|U|\ge|X|>\ell$, contradicting $|U|\le\ell$.

(N3) The first claim is Lemma~\ref{lem:R}(3). Let $x\in X$. If $z,z'\in A_X$ are non-adjacent, $z\,x\,z'$ is an induced $P_3$ with pairs $zx,xz'$ unpacked, so $zz'$ must be packed (pair-maximality). If $a\in V(H)\setminus A_X$ and $z\in A_X$ with $az\in E$, then $xa\notin E$ and $x\,z\,a$ is an induced $P_3$ with $xz,xa$ unpacked, so $az$ is packed, i.e.\ $z\in Q(a)$.

(N4) $x\in R$, so $H$ is still a packing of $G-x$, still pair-maximal (an induced $P_3$ of $G-x$ is one of $G$), and $X\setminus\{x\}$ is a class of $G-x$ with the same $A_X$. ($\Rightarrow$) Restricting a partition of $G$ to $V\setminus x$ does not increase its cost, so $\OPT(G-x)\le\OPT(G)\le|H|+\ell$. ($\Leftarrow$) Let $\Pi$ be an optimum of $G-x$ with cost $\le|H|+\ell$. Now $|X\setminus x|\ge\ell+1>\ell$, so (N2) in $G-x$ gives $\Pi(X\setminus x)=(X\setminus x)\cup A_X$. Add $x$ to this cluster: $N[x]=X\cup A_X$, so every pair at $x$ is satisfied, and the cost does not change. Hence $\OPT(G)\le|H|+\ell$. If $A_X=\emptyset$, $X$ is a clique component of $G$, which costs nothing, and $H$ is disjoint from it.

(N5) After the rules each class has at most $\ell+1$ vertices; the non-empty sets $A_X\subseteq V(H)$ are pairwise disjoint (Lemma~\ref{lem:R}), and $|V(H)|\le3|H|$. Classes with $A_X=\emptyset$ were deleted.
\end{proof}

\begin{remark}[Tightness of the threshold in (N4)]
For $\ell=0$ the threshold $|X|\ge2$ cannot be lowered to $|X|\ge1$: take the $4$-cycle of Proposition~\ref{prop:c4} below, $H=\{(0,1,3)\}$, $X=\{2\}$. Then $(G,H,0)$ is a NO instance, but $G-2$ is the induced $P_3$ $0\,1\,3$ itself, so $(G-2,H,0)$ is a YES instance. (For larger $\ell$ the notes report computer evidence of the same tightness; no proof is given here.)
\end{remark}

\begin{remark}[The rules and the user-supplied packing]\label{rem:ext}
The input packing $H$ of CEaMP need not be pair-maximal. Extend it greedily (polynomial time) to a pair-maximal $H'\supseteq H$ with $t=|H'|-|H|$ new members, set $\ell'=\ell-t$ (answer NO if $\ell'<0$, Remark~\ref{rem:max}) and apply this section to $(G,H',\ell')$. Then (N4), (N5) hold with $\ell'$ and $H'$; in particular $n\le3(\ell+2)(|H|+\ell)$ after reduction, since $|H'|\le|H|+\ell$. But $\tw(I(H'))$ may exceed $\tw(I(H))$, so these rules are a preprocessing for the parameters $(\ell,|H'|)$ or $(\ell,\tw(I(H')))$, not for $(\ell,\tw(I(H)))$ with the given $H$; and $|V(H')|$ is not bounded by $\ell$, so (N5) is not a kernel.
\end{remark}

\begin{proposition}[Partners of edits touching $R$]\label{prop:pointer}
In the setting of Corollary~\ref{cor:nf}, let $X$ be a class, $B=B_X$. Then:
\begin{enumerate}
\item[(P1)] \emph{Deletion.} If $\{x,a\}$ is an RH-deletion, $a\in A_X$, then $X$ is determined by $a$ (the unique class with $a\in A_X$): it is a unary flag at $a$ of weight $|X|$. Moreover $a$ is then cut off from every $z\in A_X\cap B$ with $z\notin Q(a)$, each such edge $az$ being an unpacked HH-deletion; so at least $|A_X\cap B|-q$ HH-edits accompany it.
\item[(P2)] \emph{No detachment.} If $B\cap A_X=\emptyset$ then $B=\emptyset$.
\item[(P3)] \emph{Insertion.} If $a\in B\setminus A_X$ then $B\cap A_X\ne\emptyset$, and for every $z\in B\cap A_X$ either $az$ is packed ($a\in Q(z)$, at most $q$ choices of $a$ per $z$) or $az$ is an unpacked non-edge, hence an inserted HH-edit.
\item[(P4)] \emph{Wide classes.} If $|A_X|\ge\ell+q$ then $B\subseteq A_X$.
\end{enumerate}
Consequently, every candidate partner $a\in V(H)$ of a class in an RH-edit is unary (P1), local to some $z$ (P3, packed case) or tied to an HH-edit (P3, second case; P1). This classifies the partners; it does not assert that the HH-edits are independent of the RH-edits.
\end{proposition}
\begin{proof}
(P1) Uniqueness is (N3). Let $a\in A_X\setminus B$ and $z\in A_X\cap B$, $z\notin Q(a)$. If $az\notin E$ then $az$ would be packed by (N3), contradicting $z\notin Q(a)$; so $az\in E$, unpacked, and $a\notin\Pi(X)\ni z$, so $az$ is deleted. At most $q$ vertices $z$ lie in $Q(a)$.

(P2) Suppose $B\ne\emptyset=B\cap A_X$. The cluster is $C=X\cup B$ (Theorem~\ref{thm:C1}), $a_X=|A_X\cap C|=0$, and by Lemma~\ref{lem:exch} moving $X$ out changes the cost by $|X|(|X|-|X|-|B|)=-|X||B|<0$, contradicting optimality.

(P3) $B\neq\emptyset$, so $B\cap A_X\ne\emptyset$ by (P2). Let $z\in B\cap A_X$. If $az\in E$ then $az$ is packed by (N3) ($a\in V(H)\setminus A_X$), i.e.\ $a\in Q(z)$. Otherwise either $az$ is packed (again $a\in Q(z)$) or it is an unpacked non-edge inside the cluster $\Pi(X)$, an insertion.

(P4) Suppose $B\not\subseteq A_X$. Put $\alpha=|A_X\cap B|$, $\delta=|A_X\setminus B|$, $\beta=|B\setminus A_X|\ge1$. By (N1) there are $|X|\beta$ RH-insertions and $|X|\delta$ RH-deletions. For $b\in B\setminus A_X$ and $z\in A_X\cap B$ with $z\notin Q(b)$, the pair $bz$ is unpacked, lies in a cluster, and is a non-edge by (N3); so it is an unpacked insertion, an HH-edit. There are at least $\beta\max(0,\alpha-q)$ of them. Hence $\ell\ge|U|\ge\beta(|X|+\max(0,\alpha-q))+|X|\delta$. If $\alpha\ge q$ this gives $\ell\ge1+(\alpha-q)+\delta$, i.e.\ $|A_X|=\alpha+\delta\le\ell+q-1$. If $\alpha<q$ it gives $\delta\le\ell-1$ and $|A_X|\le q-1+\ell-1$. Either way $|A_X|<\ell+q$.
\end{proof}

\subsection{Where locality stops}
\begin{proposition}[Strong locality fails]\label{prop:c4}
Let $G=C_4$ with edges $01,02,13,23$ and $H=\{(0,1,3)\}$ (pairs $01,13,03$; vertex $2\notin V(H)$). Then $H$ is pair-maximal, $\OPT=2=|H|+1$, so $(G,H,0)$ is a NO instance, and there are exactly three optima $\{03,12\}$, $\{01,23\}$, $\{02,13\}$, each containing an unpacked pair at vertex $2$.
\end{proposition}
This (checked exhaustively over all partitions, and verified by hand: every induced $P_3$ of $C_4$ contains one of the packed pairs $01,13,03$) refutes the claim that all unpacked edits of some optimum have both endpoints in $V(H)$; Theorem~\ref{thm:C1} is what remains true: no edit has both endpoints in $R$. In the notes' random sample ($n\le9$) about $4\%$ of maximal-packing instances with $|R|\ge2$ force an edit touching $R$ (computer evidence, not a theorem).

\begin{remark}[Vertex-disjoint packings and the case $c=1$]\label{rem:c1}
For a vertex-disjoint $H$ ($w=0$, $c=1$) FPT in $\ell$ is the theorem of~\cite{BFK18}. One must not derive it from Theorem~\ref{thm:C1} by treating $H$ as pair-maximal: a vertex-disjoint $H$ that is maximal among vertex-disjoint packings need not be pair-maximal (Proposition~\ref{prop:vertexmax}; also the path $0\,1\,2\,3\,4$ with $H=\{(0,1,2)\}$, which is extended by $(2,3,4)$ sharing the vertex $2$). The pair-maximal extension $H'$ of Remark~\ref{rem:ext} is in general no longer vertex-disjoint, so $I(H')$ has edges and the case $w=0$ is not available for it. We therefore claim nothing about $c=1$ beyond~\cite{BFK18} and Theorem~\ref{thm:K}.
\end{remark}
