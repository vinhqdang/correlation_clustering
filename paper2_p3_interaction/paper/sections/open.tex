\section{Open problems and summary of results}\label{sec:open}
Notation as above; $w=\tw(I(H))$.
\begin{problem}[Main question: $(\ell,w)$ in general; \textbf{OPEN}]\label{op:main}
Is CEaMP fixed-parameter tractable parameterized by $\ell+w$, i.e.\ solvable in $f(\ell,w)n^{O(1)}$ time, or is it W[1]-hard in $\ell+w$?
\end{problem}
\noindent Known: XP in $\ell$ with single-exponential dependence on $w$, in time $5^{w+1}n^{2\ell+O(1)}$ and $n^{\ell}f(\ell,w)5^{w+\ell+1}$ (Theorems~\ref{thm:A} and~\ref{thm:nl}); fixed-parameter tractable in $\ell$ when $w=0$~\cite{BFK18}, and in $(\ell,s)$ when the components of $I(H)$ span at most $s$ vertices (Theorem~\ref{thm:K}); fixed-parameter tractable in $(\ell,D,\omega)$ (Theorem~\ref{thm:omega}), where $\omega$ is not bounded by $(\ell,w)$ (Proposition~\ref{prop:expander}); the case $H=\emptyset$ (plain Cluster Editing, $\ell=k$) is fixed-parameter tractable. Adding $c$ to the parameter changes nothing, since $c\le w+1$. \emph{Precise status (Section~\ref{sec:soft}).} (1) Without soft triangles (Definition~\ref{def:softtri}) the set of zero-excess states is majority-closed, $\ell=0$ is polynomial for every $c$ and $w$, lock cores have at most two anchors, and the far-lift model is fixed-parameter tractable; on forests ($w\le1$) the full problem is fixed-parameter tractable in $\ell$ if the bounded-certificate hypothesis BC holds (Theorem~\ref{thm:F}), so for soft-triangle-free instances the question reduces to BC (Open Problem~\ref{op:BC}). (2) A hardness proof would have to use soft triangles, along every route we can analyse: the interior-hitting mechanism of the abstract W[1]-hard problem (Theorem~\ref{thm:B}) cannot be realised by tunnel releases without soft triangles (Theorem~\ref{thm:C}), and the only known source of unbounded certificates uses soft triangles (Theorem~\ref{thm:CE}); other operations are excluded only heuristically. Neither fixed-parameter tractability nor hardness is proved. On our route, a positive answer needs a bound $g(\ell,w)$ on the number of candidate \emph{positions} of the first endpoint of each hard edit: given one endpoint, the other has at most $Z^2$ candidates (Corollary~\ref{cor:cand}), so only the placement of at most $\ell$ bounded objects is free. Locality is proved for type T1 pairs, in particular soft-free merges (Corollary~\ref{cor:softfree}), but not for type T2 pairs or deleted edges (Propositions~\ref{prop:cex} and~\ref{prop:FAR}). The algorithm need not enumerate positions, so this is not a necessary condition.

\begin{problem}[Pointers of type T2 and deleted hard edges]\label{op:T2}\label{op:del}
Can the far ends of inserted pairs of type T2 (a kept hard edge $az$ with $a$ far from $z$ in $I(H)$, and a soft pair $zb$; by Lemma~\ref{lem:near} the end $a$ is one of at most $\ell$ vertices fixed by $(z,b)$), and of deleted hard edges between far members, be handled in a dynamic program of width bounded by $g(\ell,w)$? Proposition~\ref{prop:FAR} shows that an optimum can be forced to delete a hard edge between members at any distance in $I(H)$ (with $w=1$, $c=2$, $\ell=1$); whether a merge pointer between two pieces joined by a soft pair can be forced at unbounded distance is open.
\end{problem}
\begin{problem}[Reduction lemma; open conjecture]\label{op:ML}
Is there a polynomial-time algorithm that, given $(G,H,\ell)$ with $H$ pair-maximal, either outputs NO correctly or outputs an equivalent instance $(G^*,H^*,\ell^*)$ with $H^*$ pair-maximal, $\ell^*\le\ell$, $w(H^*)\le f_1(\ell,w)$ and $\tw(\Gamma'(G^*,H^*))\le f_2(\ell,w)$? A positive answer would give fixed-parameter tractability in $\ell+w$ by Theorem~\ref{thm:omega} (with $D^*\le2(w(H^*)+1)$). Proposition~\ref{prop:expander} shows that some reduction is necessary. For $c=1$ the rule of~\cite{BFK18} is such a reduction; Theorem~\ref{thm:three} leaves at most $2\ell$ components of $I(H)$; Appendix~\ref{app:spine} shows that the same rule applied to pair-closed unions of members removes nothing on a family with $w=1$, $\ell=1$ and unbounded $\tw(\Gamma')$.
\end{problem}
\begin{problem}[$w=1$ with unbounded tree components]\label{op:forest}
If $I(H)$ is a forest ($w\le1$, hence $c\le2$) with components of unbounded size, is CEaMP fixed-parameter tractable in $\ell$? Theorem~\ref{thm:K} settles bounded components, in particular perfect matchings (Corollary~\ref{cor:pm}), but its counting argument is at the level of whole components and gives nothing for a single large tree component. For forests there is no soft triangle: $\ell=0$ is polynomial, the far-lift model is fixed-parameter tractable (Theorem~\ref{thm:U}), and the full problem is fixed-parameter tractable in $\ell$ conditionally on BC and the oracle (O) (Theorem~\ref{thm:F}); the question is thus reduced to Open Problem~\ref{op:BC}.
\end{problem}
\begin{problem}[Packings with $c\le22$]\label{op:c2}
Is CEaMP with $c=2$ fixed-parameter tractable in $\ell$? What is the complexity of $\ell=0$ for $3\le c\le22$? Both are open in~\cite{LPS23}, whose authors say a W[1]-hardness result for $c=2$ ``would also not be surprising''. Partial answer: for soft-triangle-free packings (for $c\le2$: $I(H)$ triangle-free) $\ell=0$ is polynomial for every $c$ (Corollary~\ref{cor:sfpoly}), so a hardness proof for $\ell=0$ at $c=2$ needs triangles in $I(H)$, hence $w\ge2$. The case $c=2$ with triangles and arbitrary $\ell$ is open.
\end{problem}
\begin{problem}[Bounded certificates BC; \textbf{OPEN}]\label{op:BC}
For forests $I(H)$ (more generally soft-triangle-free packings), is there $K(t,D)$ such that every inclusion-minimal set of sites $C$ with $\mathrm{Ex}_C(G')>t$ has at most $K(t,D)$ elements (Section~\ref{sec:softmodel})? $t=0$ holds with $K=2$ (Lemma~\ref{lem:sfH2}); $t\ge1$ is open, as is the $\ell\ge1$ version of Theorem~\ref{thm:CB0} (Erd\H{o}s--P\'osa-type argument sketched, not proved). A proof needs a weighted Helly statement for five-state problems; breaks interact through locks (Proposition~\ref{prop:hub}).
\end{problem}
\begin{problem}[Soft triangles at $c\le2$]\label{op:tri2}
For $c\le2$ with triangles in $I(H)$, is $\Sigma$ always $2$-decomposable (equivalently, do lock cores have at most two anchors)? Evidence: yes for all triangle blocks, $K_4$ gadgets and random instances tried (Remark~\ref{rem:tri2}); Example~\ref{ex:maj} shows that closure under the specific operation $m_3$ fails.
\end{problem}
\begin{problem}[Small certificates]\label{op:mincert}
Can a certificate of bounded size always be chosen (or found in fixed-parameter time) when minimal certificates are unbounded (Theorem~\ref{thm:CE}, Remark~\ref{rem:mincert})?
\end{problem}
\begin{problem}[Base of the exponent]\label{op:seth}
For $\ell=0$, does a $(3-\varepsilon)^{w}n^{O(1)}$ algorithm exist, or is there a SETH-type lower bound? Theorem~\ref{thm:eth} gives only the exponent.
\end{problem}
\begin{problem}[The chain object]\label{op:tunnel}\label{op:chain}
Is there a family of tight instances with $c$ and $\tw(I(H))$ bounded independently of the size and with a core of unbounded size, in the sense of Appendix~\ref{app:tunnels}? This is the object that a W[1]-hardness proof through hard edges between far vertices (tunnels) would need. By Corollary~\ref{cor:sfcore} such a family must contain soft triangles.
\end{problem}
