\section{Open problems and summary of results}\label{sec:open}
Notation as above; $w=\tw(I(H))$.
\subsection*{The main question: answered}
The main question of this paper was whether CEaMP is fixed-parameter tractable parameterized by $\ell+w$. \textbf{Answer: no, unless $\mathrm{FPT}=\mathrm{W[1]}$} (Theorem~\ref{thm:hard}): the problem is W[1]-hard in $\ell+w$, in $\ell+\mathrm{pw}(I(H))$ and in $\ell+c$ (recall $c\le w+1$), even for pair-maximal packings, and under ETH there is no algorithm in time $f(\ell,w)N^{o(\sqrt\ell)}$ (Corollary~\ref{cor:hardeth}). The XP algorithms of Theorems~\ref{thm:A} and~\ref{thm:nl}, with exponent $2\ell$ resp.\ $\ell$ and single-exponential dependence on $w$, are therefore optimal up to the exact exponent of $n$.

\noindent Positive results that remain: fixed-parameter tractable in $\ell$ when $w=0$~\cite{BFK18}, and in $(\ell,s)$ when the components of $I(H)$ span at most $s$ vertices (Theorem~\ref{thm:K}); fixed-parameter tractable in $(\ell,D,\omega)$ (Theorem~\ref{thm:omega}), where $\omega$ is not bounded by $(\ell,w)$ (Proposition~\ref{prop:expander}) and is unbounded on the hard instances; the case $H=\emptyset$ (plain Cluster Editing, $\ell=k$) is fixed-parameter tractable. \emph{Precise status (Sections~\ref{sec:soft}, \ref{sec:hardness} and~\ref{sec:repack}).} (1) Without soft triangles (Definition~\ref{def:softtri}) the set of zero-excess states is majority-closed, $\ell=0$ is polynomial for every $c$ and $w$, lock cores have at most two anchors, and the far-lift model is fixed-parameter tractable; on forests ($w\le1$) the full problem is fixed-parameter tractable in $\ell$ if the bounded-certificate hypothesis BC holds (Theorem~\ref{thm:F}), so for soft-triangle-free instances the question reduces to BC (Open Problem~\ref{op:BC}). (2) With soft triangles the problem is W[1]-hard (Theorem~\ref{thm:hard}); the interior-hitting mechanism of the abstract problem of Theorem~\ref{thm:B} cannot be realised by tunnel releases without soft triangles (Theorem~\ref{thm:C}), and is realised with them. The hard instances have $c=4p+2\ge26$ and $w=\Theta(p^2)$. (3) \emph{Repacking.} The two known obstructions to locality, the far-deletion family FAR (Proposition~\ref{prop:FAR}) and the spine family $\mathrm{Sp}(F)$ (Proposition~\ref{prop:five}), are repacking artifacts (Propositions~\ref{prop:rpFAR} and~\ref{prop:rpSp}), a one-sided algorithm is exact on the local class (Theorem~\ref{thm:rpPL}), but locating the repacking in fixed-parameter time is impossible unless $\mathrm{FPT}=\mathrm{W[1]}$ (Corollary~\ref{cor:rpLOCno}). A positive answer to the old question would have needed a bound $g(\ell,w)$ on the number of candidate positions of the first endpoint of each hard edit; by Theorem~\ref{thm:hard} no such bound exists in general (given one endpoint, the other has at most $Z^2$ candidates, Corollary~\ref{cor:cand}, so it is the placement of at most $\ell$ bounded objects that is free, and the reduction exploits exactly this freedom).

\begin{problem}[Hardness for bounded $c$]\label{op:boundc}
Is CEaMP W[1]-hard parameterized by $\ell$ when $c$ is bounded by a constant (for instance $c\le34$)? The reduction of Theorem~\ref{thm:hard} has $c=4p+2$ with $p\ge k^2+5=\ell+5$ because a blown-up clique of size $p$ cannot be split by an edit set of excess $\ell$ only if $p>\ell$ (Lemma~\ref{lem:hsplit}); the zero-excess semantics (Proposition~\ref{prop:hTR}) hold for every even $p\ge8$. What is missing is a gadget that holds a vertex of the chain together at bounded $c$ against $\ell$ edits.
\end{problem}
\begin{problem}[ETH-tight lower bound]\label{op:eth}
Under ETH, Corollary~\ref{cor:hardeth} excludes $f(\ell,w)N^{o(\sqrt\ell)}$ (from $\ell=k^2$ and the bound $n^{o(k)}$ for Grid Tiling), while the algorithms of Theorems~\ref{thm:A} and~\ref{thm:nl} run in time $N^{2\ell+O(1)}$ resp.\ $N^{\ell+O(1)}$ times $f(\ell,w)$. Is there an algorithm with exponent $O(\sqrt\ell)$, or is $N^{o(\ell)}$ excluded? A reduction with $\ell=O(k)$ would be needed for the latter.
\end{problem}
\begin{problem}[Pointers of type T2 and deleted hard edges]\label{op:T2}\label{op:del}
Can the far ends of inserted pairs of type T2 (a kept hard edge $az$ with $a$ far from $z$ in $I(H)$, and a soft pair $zb$; by Lemma~\ref{lem:near} the end $a$ is one of at most $\ell$ vertices fixed by $(z,b)$), and of deleted hard edges between far members, be handled in a dynamic program of width bounded by $g(\ell,w)$ \emph{on a restricted class}? For the full problem the answer is no unless $\mathrm{FPT}=\mathrm{W[1]}$ (Theorem~\ref{thm:hard}), since such a program would give a fixed-parameter algorithm; the question is open for soft-triangle-free packings and bounded $c$. Proposition~\ref{prop:FAR} shows that an optimum can be forced to delete a hard edge between members at any distance in $I(H)$ (with $w=1$, $c=2$, $\ell=1$); whether a merge pointer between two pieces joined by a soft pair can be forced at unbounded distance is open. For the deleted edge of Proposition~\ref{prop:FAR} the far pointer disappears after one exchange (Proposition~\ref{prop:rpFAR}); finding such an exchange in fixed-parameter time is impossible in general unless $\mathrm{FPT}=\mathrm{W[1]}$ (Corollary~\ref{cor:rpLOCno}).
\end{problem}
\noindent\textbf{Open Problem~\ref{op:LOC}} (LOC, locating the repacking) is stated in Section~\ref{sec:repack}; by Proposition~\ref{prop:rpLOC} a positive answer implies fixed-parameter tractability in $\ell+w$, so by Theorem~\ref{thm:hard} the answer is negative unless $\mathrm{FPT}=\mathrm{W[1]}$ (Corollary~\ref{cor:rpLOCno}), and Conjecture~\ref{conj:rpC1} (normal form NF$^*$) is its existence counterpart.
\begin{problem}[Reduction lemma; negative unless $\mathrm{FPT}=\mathrm{W[1]}$]\label{op:ML}
Is there a polynomial-time algorithm that, given $(G,H,\ell)$ with $H$ pair-maximal, either outputs NO correctly or outputs an equivalent instance $(G^*,H^*,\ell^*)$ with $H^*$ pair-maximal, $\ell^*\le\ell$, $w(H^*)\le f_1(\ell,w)$ and $\tw(\Gamma'(G^*,H^*))\le f_2(\ell,w)$? A positive answer would give fixed-parameter tractability in $\ell+w$ by Theorem~\ref{thm:omega} (with $D^*\le2(w(H^*)+1)$), so by Theorem~\ref{thm:hard} it is impossible unless $\mathrm{FPT}=\mathrm{W[1]}$. Proposition~\ref{prop:expander} shows that some reduction is necessary. For $c=1$ the rule of~\cite{BFK18} is such a reduction; Theorem~\ref{thm:three} leaves at most $2\ell$ components of $I(H)$; Appendix~\ref{app:spine} shows that the same rule applied to pair-closed unions of members removes nothing on a family with $w=1$, $\ell=1$ and unbounded $\tw(\Gamma')$.
\end{problem}
\begin{problem}[$w=1$ with unbounded tree components]\label{op:forest}
If $I(H)$ is a forest ($w\le1$, hence $c\le2$) with components of unbounded size, is CEaMP fixed-parameter tractable in $\ell$? Theorem~\ref{thm:K} settles bounded components, in particular perfect matchings (Corollary~\ref{cor:pm}), but its counting argument is at the level of whole components and gives nothing for a single large tree component. For forests there is no soft triangle: $\ell=0$ is polynomial, the far-lift model is fixed-parameter tractable (Theorem~\ref{thm:U}), and the full problem is fixed-parameter tractable in $\ell$ conditionally on BC and the oracle (O) (Theorem~\ref{thm:F}); the question is thus reduced to Open Problem~\ref{op:BC}.
\end{problem}
\begin{problem}[Packings with $c\le22$]\label{op:c2}
Is CEaMP with $c=2$ fixed-parameter tractable in $\ell$? What is the complexity of $\ell=0$ for $3\le c\le22$? Both are open in~\cite{LPS23}, whose authors say a W[1]-hardness result for $c=2$ ``would also not be surprising''. Partial answer: for soft-triangle-free packings (for $c\le2$: $I(H)$ triangle-free) $\ell=0$ is polynomial for every $c$ (Corollary~\ref{cor:sfpoly}), so a hardness proof for $\ell=0$ at $c=2$ needs triangles in $I(H)$, hence $w\ge2$. The case $c=2$ with triangles and arbitrary $\ell$ is open; the reduction of Theorem~\ref{thm:hard} has $c\ge26$ (Open Problem~\ref{op:boundc}).
\end{problem}
\begin{problem}[Bounded certificates BC; \textbf{OPEN}]\label{op:BC}
For forests $I(H)$ (more generally soft-triangle-free packings), is there $K(t,D)$ such that every inclusion-minimal set of sites $C$ with $\mathrm{Ex}_C(G')>t$ has at most $K(t,D)$ elements (Section~\ref{sec:softmodel})? $t=0$ holds with $K=2$ (Lemma~\ref{lem:sfH2}); $t\ge1$ is open, as is the $\ell\ge1$ version of Theorem~\ref{thm:CB0} (Erd\H{o}s--P\'osa-type argument sketched, not proved). A proof needs a weighted Helly statement for five-state problems; breaks interact through locks (Proposition~\ref{prop:hub}).
\end{problem}
\begin{problem}[Soft triangles at $c\le2$]\label{op:tri2}
For $c\le2$ with triangles in $I(H)$, is $\Sigma$ always $2$-decomposable (equivalently, do lock cores have at most two anchors)? Evidence: yes for all triangle blocks, $K_4$ gadgets and random instances tried (Remark~\ref{rem:tri2}); Example~\ref{ex:maj} shows that closure under the specific operation $m_3$ fails.
\end{problem}
\begin{problem}[Small certificates]\label{op:mincert}
Can a certificate of bounded size always be chosen (or found in fixed-parameter time) when minimal certificates are unbounded (Theorem~\ref{thm:CE}, Remark~\ref{rem:mincert})?
\end{problem}
\begin{problem}[Base of the exponent]\label{op:seth}
For $\ell=0$, does a $(3-\varepsilon)^{w}n^{O(1)}$ algorithm exist, or is there a SETH-type lower bound? Theorem~\ref{thm:eth} gives only the exponent.
\end{problem}
