\section{Open problems}\label{sec:open}
Notation as above; $w=\tw(I(H))$.
\begin{problem}[Main question: $(\ell,w)$ in general]\label{op:main}
Is CEaMP fixed-parameter tractable parameterized by $\ell+w$, i.e.\ solvable in $f(\ell,w)n^{O(1)}$ time, or is it W[1]-hard in $\ell+w$? Known: XP in $\ell$ and FPT in $w$ for each fixed $\ell$ only via $n^{2\ell}$ (Theorem~\ref{thm:A}); FPT in $\ell$ when $w=0$ by~\cite{BFK18}, and FPT in $(\ell,s)$ when the components of $I(H)$ span at most $s$ vertices (Theorem~\ref{thm:K}); the case $H=\emptyset$ (plain Cluster Editing, $\ell=k$) is FPT. Note that adding $c$ to the parameter changes nothing, since $c\le w+1$. Also known: $n^{\ell}f(\ell,w)7^{w+\ell+1}$ time (Theorem~\ref{thm:nl}) and FPT in $(\ell,D,\omega)$ (Theorem~\ref{thm:omega}), where $\omega$ is not bounded by $(\ell,w)$ (Proposition~\ref{prop:expander}). A positive answer needs a bound $g(\ell,w)$ on the number of candidate \emph{positions} of the first endpoint of each unpacked edit: given one endpoint, the other has at most $f(\ell,D)$ candidates (Corollary~\ref{cor:cand}), so only the placement of at most $\ell$ bounded objects is free. Locality is proved for type T1 pairs, in particular soft-free merges (Corollary~\ref{cor:softfree}), but \emph{not} for type T2 pairs, which include generic merges (Proposition~\ref{prop:cex}). A negative answer has to encode $[n]$-valued information through type T2 pairs or deleted hard edges (a heuristic observation, Remark~\ref{rem:hardheur}). What exactly remains is listed in Problems~\ref{op:T2}--\ref{op:ML}.
\end{problem}
\begin{problem}[Pointers of type T2]\label{op:T2}
Can the far ends of inserted pairs of type T2 (a kept hard edge $az$ with $a$ far from $z$ in $I(H)$, and a soft pair $zb$; by Lemma~\ref{lem:near} the end $a$ is one of at most $\ell$ vertices fixed by the anchor $(z,b)$) be handled in a dynamic program of width bounded by $g(\ell,w)$? In particular, can a merge pointer between two pieces joined by a soft pair be forced at unbounded distance in $I(H)$? The computer evidence available (Remark~\ref{rem:far}) concerns only insertions inside one piece and is silent on this.
\end{problem}
\begin{problem}[Pointers realised by deleted hard edges]\label{op:del}
The same question for deleted hard edges between far members of $H$ (the other source of non-locality identified in Section~\ref{sec:left}). Given one endpoint, the other is one of at most $D+\ell$ hard neighbours (Corollary~\ref{cor:cand}); the open point is the position of the first endpoint and the interaction of the far ends with the rest of the solution.
\end{problem}
\begin{problem}[The reduction lemma ML]\label{op:ML}
Is Conjecture~\ref{conj:ML} true: is there a polynomial-time reduction after which the treewidth $\omega$ of the graph $\Gamma'$ of Theorem~\ref{thm:omega} is bounded by a function of $(\ell,w)$? A positive answer would give FPT in $\ell+w$ (Observation~\ref{obs:cond}). The family of Proposition~\ref{prop:expander} shows that a reduction is necessary; for $c=1$ the rule of~\cite{BFK18} is such a reduction, for $w\ge1$ it is unknown.
\end{problem}
\begin{problem}[$w=1$ with unbounded tree components]\label{op:forest}
If $I(H)$ is a forest ($w\le1$, hence $c\le2$) with components of unbounded size, is CEaMP FPT in $\ell$? Theorem~\ref{thm:K} settles bounded components, in particular perfect matchings (Corollary~\ref{cor:pm}), but its counting argument is at the level of whole components and gives nothing for a single large tree component.
\end{problem}
\begin{problem}[Half-integral and intermediate packings]\label{op:c2}
Is CEaMP with $c=2$ FPT in $\ell$? What is the complexity of $\ell=0$ for $3\le c\le22$? Both are open in~\cite{LPS23}, whose authors say a W[1]-hardness result for $c=2$ ``would also not be surprising''.
\end{problem}
\begin{problem}[Base of the exponent]\label{op:seth}
For $\ell=0$, does a $(3-\varepsilon)^{w}n^{O(1)}$ (or even $2^{o(w)}$ with a smaller base) algorithm exist, or is there a SETH-type lower bound? Theorem~\ref{thm:eth} gives only the exponent.
\end{problem}
\begin{problem}[Treewidth of $G$]\label{op:twg}
What is the complexity of CEaMP with $\ell=0$ (or of Cluster Editing) when $\tw(G)$ is bounded? We found no published result on Cluster Editing parameterized by treewidth alone (Section~\ref{sec:related}), so this must be proved rather than cited.
\end{problem}
